{# A checkbox dropdown for a multi-value field (0.5.93 Users page, everywhere since 0.5.94): `options` are (value, text) pairs, `selected` the current values. The hidden empty field makes "none chosen" reach the API as an empty list. `single=True` makes it a radio dropdown (one value, no hidden field) so single choices look the same. #} {% macro multi(name, options, selected, id, label='', empty='None', single=False) %} {% set chosen = [] %}{% for v, text in options %}{% if v in selected %}{% set _ = chosen.append(text) %}{% endif %}{% endfor %}
{% if label %}{{ label }}{% endif %}
{{ chosen|join(', ') or selected|join(', ') or empty }}
{% if not single %}{% endif %}{% for v, text in options %}{% else %}Nothing to pick yet{% endfor %}
{% endmacro %}
